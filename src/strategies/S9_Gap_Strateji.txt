// STRATEJİ: S9 GAP MOMENTUM (Görsel Backtest Kodu)
// SADECE "GAP YÖNÜNE DEVAM" (MOMENTUM) İŞLEMLERİNİ HESAPLAR
// ===============================================================================================
var Sembol = "VIP'VIP-X030-T";
var V = Sistem.GrafikVerileri;
var O = Sistem.GrafikFiyatOku(V, "Acilis");
var H = Sistem.GrafikFiyatOku(V, "Yuksek");
var L = Sistem.GrafikFiyatOku(V, "Dusuk");
var C = Sistem.GrafikFiyatOku(V, "Kapanis");

var Sinyal = "";
var SonYon = "";
var GercekFiyat = Sistem.Liste(0); // Performans paneline doğru fiyatı iletmek için


// --- PARAMETRELER (S9) ---
double MinGapPuan   = 9.0;
double MaxGapPuan   = 200.0;
double S9_Izleyen_Stop_Yuzde = 1.20; // Fiyat zirveden %1.2 duserse stop
double S9_Kar_Al_Yuzde = 0.0;        // 0 = Sadece Izleyen Stop calissin
int OrBars = 1;                      // OR bar süresi (09:30)

// --- DURUM DEĞİŞKENLERİ ---
int in_pos = 0; // 1 = long, -1 = short
double entry_price = 0;
double stop_level = 0;
double t1_lvl = 0;
double t2_lvl = 0;
bool t1_hit = false;
double extreme_price = 0;

bool gap_active = false;
int gap_dir = 0; // 1 = UP, -1 = DOWN
double or_high = 0;
double or_low = 999999;
double or_range = 0;
bool or_complete = false;
int or_start_bar = -1;
double last_aksam_close = 0;
double gap_fill_lvl = 0;
bool prev_is_gun = false;
int pos_start_bar = -1;

// --- GERÇEK GETİRİ HESAPLAYICI ---
double TotalPuan = 0;
int GercekToplamIslem = 0;
int GercekKarlar = 0;
int GercekZararlar = 0;

for (int i = 1; i < V.Count; i++)
{
    var dt = V[i].Date;
    double th = dt.Hour + dt.Minute / 60.0;
    
    bool is_aksam = th >= 19.0 && th <= 22.99;
    bool is_gun = th >= 9.5 && th <= 18.25;
    bool is_first_gun_bar = is_gun && !prev_is_gun;
    prev_is_gun = is_gun;
    
    if (is_aksam && dt.Hour >= 22 && dt.Minute >= 55) {
        last_aksam_close = C[i];
    }
    
    // 1. GAP TESPİTİ (09:30 BARI AÇILIŞIYLA - PYTHON İLE BİREBİR)
    if (is_first_gun_bar)
    {
        if (in_pos != 0)
        {
            double sabah_cikis = O[i];
            double islemPuan = in_pos == 1 ? (sabah_cikis - entry_price) : (entry_price - sabah_cikis);
            double son_lot = t1_hit ? 2.0 : 1.0;
            TotalPuan += (islemPuan * son_lot);
            GercekToplamIslem++;
            if (islemPuan > 0) GercekKarlar++; else GercekZararlar++;
            in_pos = 0;
            Sinyal = "F";
            GercekFiyat[i] = (float)sabah_cikis;
        }

        double ref_close = last_aksam_close > 0 ? last_aksam_close : C[i - 1];
        double raw_gap = O[i] - ref_close; // 09:30 Açılışı
        double gap_abs = Math.Abs(raw_gap);
        
        if (gap_abs >= MinGapPuan && gap_abs <= MaxGapPuan)
        {
            gap_active = true;
            gap_dir = raw_gap > 0 ? 1 : -1;
            or_start_bar = i; // 09:30 barı
            or_high = H[i];
            or_low = L[i];
            or_complete = false;
        }
        else
        {
            gap_active = false;
        }
        last_aksam_close = 0;
    }
    
    // 2. OR OLUŞTURMA (09:30 BARINDA ÇALIŞIR, 09:31'DE TAMAMLANIR)
    if (gap_active && !or_complete && is_gun)
    {
        int elapsed_or = i - or_start_bar;
        if (elapsed_or < OrBars)
        {
            if (H[i] > or_high) or_high = H[i];
            if (L[i] < or_low) or_low = L[i];
        }
        else
        {
            or_complete = true;
            or_range = or_high - or_low;
        }
    }
    
    // 3. İŞLEME GİRİŞ KONTROLÜ (09:31 VE SONRASI)
    
    // GİRİŞ KONTROLLERİ (S9 MOMENTUM)
    
    if (gap_active && or_complete && in_pos == 0)
    {
        int elapsed_since_or = i - (or_start_bar + OrBars);
        bool s9_zaman = elapsed_since_or < 300; 
        
        if (s9_zaman && or_range >= 1.0) // Düz bar filtresi
        {
            if (gap_dir == 1) // UP GAP -> S9 LONG arar (OR_High yukarı kırılırsa)
            {
                if (L[i] <= or_low) { gap_active = false; } // Reversal çalıştıysa S9 iptal
                else if (H[i] >= or_high)
                {
                    in_pos = 1;
                    Sinyal = "A";
                    pos_start_bar = i;
                    entry_price = Math.Max(O[i], or_high); // Kötümser Giriş
                    GercekFiyat[i] = (float)entry_price;
                    extreme_price = entry_price;
                }
            }
            else if (gap_dir == -1) // DOWN GAP -> S9 SHORT arar (OR_Low aşağı kırılırsa)
            {
                if (H[i] >= or_high) { gap_active = false; } // Reversal çalıştıysa S9 iptal
                else if (L[i] <= or_low)
                {
                    in_pos = -1;
                    Sinyal = "S";
                    pos_start_bar = i;
                    entry_price = Math.Min(O[i], or_low); // Kötümser Giriş
                    GercekFiyat[i] = (float)entry_price;
                    extreme_price = entry_price;
                }
            }
        }
    }
    
    // TAKİP KONTROLLERİ (S9 İÇİN) - YÜZDESEL İZLEYEN STOP
    if (in_pos == 1)
    {
        if (C[i] > extreme_price) extreme_price = C[i];
        
        bool kar_al_hit = (S9_Kar_Al_Yuzde > 0) && (C[i] >= entry_price * (1 + S9_Kar_Al_Yuzde / 100.0));
        bool stop_hit = (S9_Izleyen_Stop_Yuzde > 0) && (C[i] <= extreme_price * (1 - S9_Izleyen_Stop_Yuzde / 100.0));
        
        bool zaman_doldu = pos_start_bar > 0 && (i - (or_start_bar+OrBars)) >= 300;
        bool aksam_kapa = is_aksam && th >= (22 + 50.0 / 60.0);
        
        if (kar_al_hit || stop_hit || zaman_doldu || aksam_kapa) {
            Sinyal = "F";
            in_pos = 0;
            GercekFiyat[i] = (float)C[i];
            
            double islemPuan = (C[i] - entry_price);
            TotalPuan += islemPuan;
            GercekToplamIslem++;
            if (islemPuan > 0) GercekKarlar++; else GercekZararlar++;
            
            gap_active = false;
        }
    }
    else if (in_pos == -1)
    {
        if (extreme_price == 0 || C[i] < extreme_price) extreme_price = C[i];
        
        bool kar_al_hit = (S9_Kar_Al_Yuzde > 0) && (C[i] <= entry_price * (1 - S9_Kar_Al_Yuzde / 100.0));
        bool stop_hit = (S9_Izleyen_Stop_Yuzde > 0) && (C[i] >= extreme_price * (1 + S9_Izleyen_Stop_Yuzde / 100.0));
        
        bool zaman_doldu = pos_start_bar > 0 && (i - (or_start_bar+OrBars)) >= 300;
        bool aksam_kapa = is_aksam && th >= (22 + 50.0 / 60.0);
        
        if (kar_al_hit || stop_hit || zaman_doldu || aksam_kapa) {
            Sinyal = "F";
            in_pos = 0;
            GercekFiyat[i] = (float)C[i];
            
            double islemPuan = (entry_price - C[i]);
            TotalPuan += islemPuan;
            GercekToplamIslem++;
            if (islemPuan > 0) GercekKarlar++; else GercekZararlar++;
            
            gap_active = false;
        }
    }
    
    if (Sinyal != "" && SonYon != Sinyal) {
        SonYon = Sinyal;
        Sistem.Yon[i] = SonYon;
    }
}

{

    {
        float gercekFiy = (GercekFiyat[i] > 0) ? GercekFiyat[i] : C[i];

        if (Sistem.Yon[i] == "A") {
            if (poz < 0) {
                anlikKapananIslemKari = (poz == -1) ? (entry1 - gercekFiy - kayma) : ((entry1 - gercekFiy - kayma) + (entry2 - gercekFiy - kayma));
                kapananKZ += anlikKapananIslemKari; islemKapandi = true;
            }
            poz = 1;
            entry1 = gercekFiy + kayma;
        }
        else if (Sistem.Yon[i] == "AEK" && poz == 1) {
            poz = 2;
            entry2 = gercekFiy + kayma;
        }
        else if (Sistem.Yon[i] == "S") {
            if (poz > 0) {
                anlikKapananIslemKari = (poz == 1) ? (gercekFiy - entry1 - kayma) : ((gercekFiy - entry1 - kayma) + (gercekFiy - entry2 - kayma));
                kapananKZ += anlikKapananIslemKari; islemKapandi = true;
            }
            poz = -1;
            entry1 = gercekFiy - kayma;
        }
        else if (Sistem.Yon[i] == "SEK" && poz == -1) {
            poz = -2;
            entry2 = gercekFiy - kayma;
        }
        else if (Sistem.Yon[i] == "F" && poz != 0) {
            if (poz == 1) anlikKapananIslemKari = (gercekFiy - entry1 - kayma);
            else if (poz == 2) anlikKapananIslemKari = (gercekFiy - entry1 - kayma) + (gercekFiy - entry2 - kayma);
            else if (poz == -1) anlikKapananIslemKari = (entry1 - gercekFiy - kayma);
            else if (poz == -2) anlikKapananIslemKari = (entry1 - gercekFiy - kayma) + (entry2 - gercekFiy - kayma);
            
            kapananKZ += anlikKapananIslemKari;
            islemKapandi = true;
            poz = 0;
        }

        if (islemKapandi) {
            IslemGetirileri.Add(anlikKapananIslemKari);
            if (anlikKapananIslemKari > 0) { ToplamKar += anlikKapananIslemKari; KarliIslemSayisi++; }
            else if (anlikKapananIslemKari < 0) { ToplamZarar += Math.Abs(anlikKapananIslemKari); ZararliIslemSayisi++; }
        }

        float acikKZ = 0;
        if (poz == 1) acikKZ = C[i] - entry1;
        else if (poz == 2) acikKZ = (C[i] - entry1) + (C[i] - entry2);
        else if (poz == -1) acikKZ = entry1 - C[i];
        else if (poz == -2) acikKZ = (entry1 - C[i]) + (entry2 - C[i]);

        SanalGetiri[i] = kapananKZ + acikKZ;

        if (SanalGetiri[i] > ZirveBakiye) ZirveBakiye = SanalGetiri[i];
        float anlikDD = ZirveBakiye - SanalGetiri[i];
        if (anlikDD > GercekMaxDD) { GercekMaxDD = anlikDD; GercekMaxDDTarih = Grafikler[i].Date; }
    }
}
