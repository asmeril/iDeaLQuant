// ===============================================================================================
// STRATEJİ: S8 GAP REVERSAL (Görsel Backtest Kodu)
// ===============================================================================================
// Periyot   : 1 Dakika
// Vade Tipi : ENDEKS
//
// NOT: Önceki versiyondaki OR (Açılış Aralığı) zamanlama hatası düzeltildi! 
// 09:25 emir toplama barı yerine, gerçek piyasa açılışı olan ilk 09:30 barı 
// Opening Range olarak referans alınmıştır.
// ===============================================================================================

var V = Sistem.GrafikVerileri;
var O = Sistem.GrafikFiyatSec("Acilis");
var H = Sistem.GrafikFiyatSec("Yuksek");
var L = Sistem.GrafikFiyatSec("Dusuk");
var C = Sistem.GrafikFiyatSec("Kapanis");

for (int i = 0; i < Sistem.BarSayisi; i++) Sistem.Yon[i] = "";

// --- PARAMETRELER (Yeni Optimizasyon Sonuçları) ---
// Not: İki strateji aynı gap filtlerini kullandığı için ortak bir denge seçildi (10 - 150)
double MinGapPuan   = 10.0;     // Ortak Denge (S9: 8, S8: 21)
double MaxGapPuan   = 150.0;    // Ortak Denge
double S8_Izleyen_Stop_Yuzde = 1.5; // S8 Fiyat zirveden %1.5 duserse stop
double S8_Kar_Al_Yuzde = 1.0;       // S8 Fiyat %1.0 kar yaparsa direkt cikar
int OrBars          = 1;        // OR bar s resi (09:30 bar )

// --- DURUM DEĞİŞKENLERİ  ---
var GercekFiyat = Sistem.Liste(0); // Gercek Fiyatlari tutmak icin
int in_pos = 0; // 1 = long, -1 = short
double entry_price = 0;
double t1_lvl = 0;
double t2_lvl = 0;
bool t1_hit = false;
double extreme_price_s8 = 0;
bool is_s8 = false;

bool gap_active = false;
int gap_dir = 0;
double or_high = 0;
double or_low = float.MaxValue;
bool or_complete = false;
int or_start_bar = -1;
double gap_fill_lvl = 0;

double last_aksam_close = 0;
bool prev_is_gun = false;

string SonYon = "";

for (int i = 1; i < Sistem.BarSayisi; i++)
{
    var dt = V[i].Date;
    double th = dt.Hour + dt.Minute / 60.0;
    string Sinyal = "";
    
    bool is_aksam = 19.0 <= th && th <= 23.0;
    bool is_gun = (9 + 30.0 / 60.0) <= th && th <= (18 + 15.0 / 60.0);
    bool is_first_gun_bar = is_gun && !prev_is_gun;
    prev_is_gun = is_gun;
    
    // Gece kapanışını kaydet
    if (is_aksam && dt.Hour >= 22 && dt.Minute >= 55) {
        last_aksam_close = C[i];
    }
    
    // Yeni günün İLK işlem barı (09:30)
    if (is_first_gun_bar) 
    {
        //  Önceki günden pozisyon kaldıysa zorunlu kapat (Safety)
        if (in_pos != 0) {
            double sabah_cikis = O[i];
            in_pos = 0;
            Sinyal = "F"; 
            GercekFiyat[i] = (float)sabah_cikis;
        }
        
        double ref_close = last_aksam_close > 0 ? last_aksam_close : C[i - 1];
        double raw_gap = O[i] - ref_close; // 09:30 Açılışı ile dünkü kapanış arası fark
        double gap_abs = Math.Abs(raw_gap);
        
        if (gap_abs >= MinGapPuan && gap_abs <= MaxGapPuan) {
            gap_active = true;
            gap_dir = raw_gap > 0 ? 1 : -1;
            gap_fill_lvl = ref_close;
            or_start_bar = i;
            or_high = H[i];
            or_low = L[i];
            or_complete = false;
        } else {
            gap_active = false;
        }
        last_aksam_close = 0;
    }
    
    // OR oluşumu (09:30 barı ve gerekirse sonrası)
    if (gap_active && !or_complete && is_gun) {
        int elapsed = i - or_start_bar;
        if (elapsed < OrBars) {
            if (H[i] > or_high) or_high = H[i];
            if (L[i] < or_low) or_low = L[i];
        } else {
            or_complete = true;
        }
    }
    
    // Giriş (Entry) Mantığı (09:31 ve sonrası)
    if (gap_active && or_complete && in_pos == 0) 
    {
        int or_end_bar = or_start_bar + OrBars;
        bool s8_zaman_ok = (i - or_end_bar) < 120; // S8 zaman limiti (~11:30)
                double or_range = or_high - or_low;
        
        if (gap_dir == 1) // YUKARI GAP
        {
            // S8 Ters Yön Kırılım (Short)
            if (s8_zaman_ok && L[i] <= or_low && H[i] > gap_fill_lvl) {
                in_pos = -1;
                is_s8 = true;
                entry_price = Math.Min(O[i], or_low);
                extreme_price_s8 = entry_price;
                GercekFiyat[i] = (float)entry_price;
                Sinyal = "S";
            }
        }
        else if (gap_dir == -1) // AŞAĞI GAP
        {
            // S8 Ters Yön Kırılım (Long)
            if (s8_zaman_ok && H[i] >= or_high && L[i] < gap_fill_lvl) {
                in_pos = 1;
                is_s8 = true;
                entry_price = Math.Max(O[i], or_high);
                extreme_price_s8 = entry_price;
                GercekFiyat[i] = (float)entry_price;
                Sinyal = "A";
            }
        }
    }
    
    //   Çıkış (Exit) Mantığı
    if (in_pos == 1 && Sinyal == "") 
    {
        if (is_s8) {
            if (H[i] > extreme_price_s8) extreme_price_s8 = H[i];
            
            double tp_lvl = entry_price * (1 + S8_Kar_Al_Yuzde / 100.0);
            double sl_lvl = extreme_price_s8 * (1 - S8_Izleyen_Stop_Yuzde / 100.0);
            
            bool kar_al_hit = (S8_Kar_Al_Yuzde > 0) && (H[i] >= tp_lvl);
            bool stop_hit = (S8_Izleyen_Stop_Yuzde > 0) && (L[i] <= sl_lvl);
            
            bool zaman_doldu = (i - (or_start_bar + OrBars)) >= 120;
            bool aksam_kapa = is_aksam && th >= (22 + 50.0 / 60.0);
            bool gun_sonu = th >= (18 + 8.0 / 60.0) && th <= (18 + 10.0 / 60.0);
            
            if (kar_al_hit || stop_hit || zaman_doldu || aksam_kapa || gun_sonu) {
                Sinyal = "F";
                in_pos = 0;
                gap_active = false;
                GercekFiyat[i] = (float)(kar_al_hit ? tp_lvl : (stop_hit ? sl_lvl : C[i]));
            }
        }
    }
    else if (in_pos == -1 && Sinyal == "") 
    {
        if (is_s8) {
            if (extreme_price_s8 == 0 || L[i] < extreme_price_s8) extreme_price_s8 = L[i];
            
            double tp_lvl = entry_price * (1 - S8_Kar_Al_Yuzde / 100.0);
            double sl_lvl = extreme_price_s8 * (1 + S8_Izleyen_Stop_Yuzde / 100.0);
            
            bool kar_al_hit = (S8_Kar_Al_Yuzde > 0) && (L[i] <= tp_lvl);
            bool stop_hit = (S8_Izleyen_Stop_Yuzde > 0) && (H[i] >= sl_lvl);
            
            bool zaman_doldu = (i - (or_start_bar + OrBars)) >= 120;
            bool aksam_kapa = is_aksam && th >= (22 + 50.0 / 60.0);
            bool gun_sonu = th >= (18 + 8.0 / 60.0) && th <= (18 + 10.0 / 60.0);
            
            if (kar_al_hit || stop_hit || zaman_doldu || aksam_kapa || gun_sonu) {
                Sinyal = "F";
                in_pos = 0;
                gap_active = false;
                GercekFiyat[i] = (float)(kar_al_hit ? tp_lvl : (stop_hit ? sl_lvl : C[i]));
            }
        }
    }
    
    if (Sinyal != "" && SonYon != Sinyal) {
        SonYon = Sinyal;
        Sistem.Yon[i] = SonYon;
    }
}

bool DetayPerformans = true;

// 1. ADIM: GERÇEK PİRAMİT MATEMATİĞİ İLE GETİRİ EĞRİSİ
var SanalGetiri = Sistem.Liste(0);

for (int i = 1; i < Grafikler.Count; i++)
{
    float anlikKapananIslemKari = 0;
    bool islemKapandi = false;

    if (Grafikler[i].Date >= dateBaslangicTarih)
    {
        float gercekFiy = (GercekFiyat[i] > 0) ? GercekFiyat[i] : C[i];

        if (Sistem.Yon[i] == "A") {
            if (poz < 0) {
                anlikKapananIslemKari = (poz == -1) ? (entry1 - gercekFiy - kayma) : ((entry1 - gercekFiy - kayma) + (entry2 - gercekFiy - kayma));
                kapananKZ += anlikKapananIslemKari; islemKapandi = true;
            }
            poz = 1;
            entry1 = gercekFiy + kayma;
        }
        else if (Sistem.Yon[i] == "AEK" && poz == 1) {
            poz = 2;
            entry2 = gercekFiy + kayma;
        }
        else if (Sistem.Yon[i] == "S") {
            if (poz > 0) {
                anlikKapananIslemKari = (poz == 1) ? (gercekFiy - entry1 - kayma) : ((gercekFiy - entry1 - kayma) + (gercekFiy - entry2 - kayma));
                kapananKZ += anlikKapananIslemKari; islemKapandi = true;
            }
            poz = -1;
            entry1 = gercekFiy - kayma;
        }
        else if (Sistem.Yon[i] == "SEK" && poz == -1) {
            poz = -2;
            entry2 = gercekFiy - kayma;
        }
        else if (Sistem.Yon[i] == "F" && poz != 0) {
            if (poz == 1) anlikKapananIslemKari = (gercekFiy - entry1 - kayma);
            else if (poz == 2) anlikKapananIslemKari = (gercekFiy - entry1 - kayma) + (gercekFiy - entry2 - kayma);
            else if (poz == -1) anlikKapananIslemKari = (entry1 - gercekFiy - kayma);
            else if (poz == -2) anlikKapananIslemKari = (entry1 - gercekFiy - kayma) + (entry2 - gercekFiy - kayma);
            
            kapananKZ += anlikKapananIslemKari;
            islemKapandi = true;
            poz = 0;
        }

        if (islemKapandi) {
            IslemGetirileri.Add(anlikKapananIslemKari);
            if (anlikKapananIslemKari > 0) { ToplamKar += anlikKapananIslemKari; KarliIslemSayisi++; }
            else if (anlikKapananIslemKari < 0) { ToplamZarar += Math.Abs(anlikKapananIslemKari); ZararliIslemSayisi++; }
        }

        float acikKZ = 0;
        if (poz == 1) acikKZ = C[i] - entry1;
        else if (poz == 2) acikKZ = (C[i] - entry1) + (C[i] - entry2);
        else if (poz == -1) acikKZ = entry1 - C[i];
        else if (poz == -2) acikKZ = (entry1 - C[i]) + (entry2 - C[i]);

        SanalGetiri[i] = kapananKZ + acikKZ;

        if (SanalGetiri[i] > ZirveBakiye) ZirveBakiye = SanalGetiri[i];
        float anlikDD = ZirveBakiye - SanalGetiri[i];
        if (anlikDD > GercekMaxDD) { GercekMaxDD = anlikDD; GercekMaxDDTarih = Grafikler[i].Date; }
    }
}

    }
}
float lastDaily = SanalGetiri[Grafikler.Count-1] - prevDayEnd;
if (lastDaily > 0) MutluGun++; else if (lastDaily < 0) MutsuzGun++;

// Günlük Hesaplamalar
var DateBugun = new DateTime(DateTime.Now.Year, DateTime.Now.Month, DateTime.Now.Day);
var DateDunSonBarNo = 0;
for (int i = Grafikler.Count - 1; i > 0; i--) { if (Grafikler[i].Date < DateBugun) { DateDunSonBarNo = i; break; } }

if (Sistem.Parametreler[3] == "X")
{
    int ilksatirY = 33;

    var kzSure_yuzde = "  %" + yuzde_kz.ToString("0.0");

    string ToplamIslem = ToplamIslemSayisi.ToString("0");
    string OrtalamaIslem = (Sure > 0) ? (ToplamIslemSayisi / Sure).ToString("0") : "0";
    var KarliIslemOran = (ToplamIslemSayisi > 0) ? (((float)KarliIslemSayisi / ToplamIslemSayisi) * 100).ToString("0.00") : "0.00";
    var ProfitFactor = (ToplamZarar > 0) ? (ToplamKar / ToplamZarar).ToString("0.00") : "99.9";

    var MaxDD = GercekMaxDD.ToString("0.0");
    var MaxDDTarihi = (GercekMaxDDTarih != DateTime.MinValue) ? GercekMaxDDTarih.ToString("dd.MM.yyyy") : "-";
